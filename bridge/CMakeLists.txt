cmake_minimum_required(VERSION 3.20)
project(MaxTrackBridge LANGUAGES CXX)
add_library(MaxTrackBridge SHARED bridge.cpp)
target_include_directories(MaxTrackBridge PRIVATE ../vendor/scs_sdk/include)
target_compile_features(MaxTrackBridge PRIVATE cxx_std_17)
target_link_libraries(MaxTrackBridge PRIVATE ws2_32)
set_target_properties(MaxTrackBridge PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
enable_testing()
add_executable(test_bridge test_bridge.cpp)
target_include_directories(test_bridge PRIVATE ../vendor/scs_sdk/include)
target_compile_features(test_bridge PRIVATE cxx_std_17)
target_link_libraries(test_bridge PRIVATE ws2_32)
if(MSVC)
  target_compile_options(test_bridge PRIVATE /UNDEBUG)
else()
  target_compile_options(test_bridge PRIVATE -UNDEBUG)
endif()
add_test(NAME bridge_transport COMMAND test_bridge)
